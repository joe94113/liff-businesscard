/* global vcard */
const DEFAULT_LINK = 'https://taichunmin.idv.tw/liff-businesscard/'
function vcardUuid (vcard, secret) {
  const hash = CryptoJS.HmacMD5(JSON.stringify(vcard), secret)
  const hex = CryptoJS.enc.Hex.stringify(hash)
  return hex.replace(/^(.{8})(.{4})(.{4})(.{4})(.{12})$/, '$1-$2-$3-$4-$5')
}
function gaScreenView (vcard) {
  const CARD_NAME = '多頁訊息1-2021/06/20'
  return `https://www.google-analytics.com/collect?${Qs.stringify({
    an: 'LINE 數位版名片',
    cd: `${CARD_NAME}-${_.get(vcard, 'cards.0.title')}`, // 畫面名稱，在此用來紀錄數位版名片的版本
    cid: vcardUuid(vcard, CARD_NAME), // uuid
    ds: 'app', // 資料來源，填寫為 app
    sc: 'start',
    t: 'screenview',
    tid: 'UA-39556213-12', // GA 追蹤代碼
    ul: 'zh-tw', // locale
    v: 1, // api version
  })}`
}
function renderBtn(ctx) {
  const { btn, vcard } = ctx
  const uri = btn.link || DEFAULT_LINK
  return {
    color: btn.color || (btn.style === 'primary' ? '#388e3c' : '#e0e0e0'),
    height: btn.style === 'primary' ? 'md' : 'sm',
    style: btn.style || 'primary',
    margin: btn.style === 'link' ? 'md' : 'none',
    type: 'button',
    action: {
      label: btn.text || '預設按鈕文字',
      type: 'uri',
      uri,
    },
  }
}
function renderCard(ctx) {
  const { card, vcard } = ctx
  const uri = card.link || DEFAULT_LINK

  // 結構化資訊欄位
  const infoRows = [
    { icon: '📍', label: '地點', text: card.address || '', color: '#388e3c' },
    { icon: '🕒', label: '時間', text: card.time || '', color: '#1976d2' },
    { icon: '👥', label: '人數', text: card.people || '', color: '#8d6e63' },
    { icon: '💰', label: '費用', text: card.fee || '', color: '#d84315' }
  ].filter(row => row.text);

  // fallback: desc 每行顯示
  let infoContents = infoRows.length > 0
    ? infoRows.map(row => ({
        type: 'box',
        layout: 'horizontal',
        spacing: 'sm',
        contents: [
          { type: 'text', text: row.icon, size: 'md', flex: 0, color: row.color },
          { type: 'text', text: `${row.label}：${row.text}`, size: vcard.descSize || 'sm', color: row.color, flex: 4, wrap: true }
        ],
        margin: 'sm',
      }))
    : (card.desc
        ? card.desc.split('\n').map(line => ({
            type: 'text',
            text: line,
            size: vcard.descSize || 'sm',
            color: card.descColor || '#555555',
            wrap: true,
            margin: 'sm',
          }))
        : []);

  return {
    type: 'bubble',
    size: 'mega',
    hero: {
      animated: true,
      aspectMode: 'cover',
      aspectRatio: vcard.ratio || '20:13',
      size: 'full',
      type: 'image',
      url: card.image || '',
      action: {
        type: 'uri',
        uri,
      },
    },
    body: {
      backgroundColor: card.bgColor || '#f5f5f5',
      layout: 'vertical',
      spacing: 'md',
      type: 'box',
      contents: [
        {
          type: 'text',
          text: card.title,
          weight: 'bold',
          size: vcard.titleSize || 'xl',
          wrap: true,
          color: card.titleColor || '#1b5e20',
          margin: 'none',
        },
        {
          type: 'separator',
          margin: 'md'
        },
        ...infoContents,
        {
          type: 'separator',
          margin: 'md'
        },
      ],
    },
    footer: {
      backgroundColor: '#e8f5e9',
      layout: 'horizontal',
      spacing: 'sm',
      type: 'box',
      contents: _.map(card.btns, btn => renderBtn({ ...ctx, btn })),
    },
  }
}
function renderCarousel (ctx) {
  const { vcard } = ctx
  return {
    type: 'flex',
    altText: vcard.altText,
    contents: {
      type: 'carousel',
      contents: _.map(vcard.cards, (card, cardIdx) => renderCard({ ...ctx, card, cardIdx })),
    },
  }
}
